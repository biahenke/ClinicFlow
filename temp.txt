        async function carregarAgendaDoDia(dataAtual) {
            const token = localStorage.getItem('token');
            const agendaBody = document.getElementById('agenda-list-body');
            const consultasHojeEl = document.getElementById('stat-consultas');
            
            agendaBody.innerHTML = '<div style="text-align: center; color: rgba(255,255,255,0.7); font-size: 0.875rem;">Carregando...</div>';

            try {
                // Fetch com o parametro data para trazer apenas as da data e ja ordenadas asc
                const response = await fetch(`http://localhost:8000/api/consultas/?skip=0&limit=1000&data=${dataAtual}`, {
                    headers: { 'Authorization': `Bearer ${token}` }
                });
                
                if(!response.ok) throw new Error('Erro na resposta');
                const data = await response.json();
                
                const consultasHoje = data.items;
                
                // Filter out canceled appointments for the dashboard count
                const countNaoCanceladas = consultasHoje.filter(c => !["cancelada", "canceladas", "nao realizada", "falta"].includes(c.status?.toLowerCase())).length;
                
                if (consultasHojeEl) consultasHojeEl.textContent = countNaoCanceladas;

                agendaBody.innerHTML = '';
                if (consultasHoje.length === 0) {
                    agendaBody.innerHTML = '<div style="text-align: center; color: rgba(255,255,255,0.7); font-size: 0.875rem;">Nenhuma consulta para este dia.</div>';
                } else {
                    consultasHoje.forEach(c => {
                        const div = document.createElement('div');
                        div.className = 'agenda-item';
                        div.innerHTML = `
                            <div class="agenda-time">${c.horario.substring(0,5)}</div>
                            <div class="agenda-details">
                                <h5>${c.paciente.user.nome}</h5>
                                <p>${c.medico.user.nome} - ${c.medico.especialidade}</p>
                            </div>
                        `;
                        agendaBody.appendChild(div);
                    });
                }
            } catch (error) {
                console.error(error);
                agendaBody.innerHTML = '<div style="text-align: center; color: #F56565; font-size: 0.875rem;">Erro ao buscar agenda.</div>';
                if (consultasHojeEl) consultasHojeEl.textContent = "--";
            }
        }

        async function carregarGraficoSemanaEStatus() {
            const token = localStorage.getItem('token');
            const chartContainer = document.getElementById('grafico-semana');
            if(!chartContainer) return;
            
            try {
                // Gr├ífico da Semana
                const response = await fetch(`http://localhost:8000/api/consultas/estatisticas/semana`, {
                    headers: { 'Authorization': `Bearer ${token}` }
                });
                
                if(response.ok) {
                    const data = await response.json();
                    const valores = data.dados_semana;
                    
                    const maxVal = Math.max(...valores, 1);
                    chartContainer.innerHTML = '';
                    
                    const hojeDiaSemana = new Date().getDay(); 
                    const indexHoje = hojeDiaSemana - 1; 
                    
                    valores.forEach((val, index) => {
                        const perc = (val / maxVal) * 100;
                        const cssClass = index === indexHoje ? 'chart-bar' : 'chart-bar secondary';
                        chartContainer.innerHTML += `
                            <div class="${cssClass}" style="height: ${perc}%; position: relative;" title="${val} consultas">
                                <span style="position: absolute; top: -20px; left: 50%; transform: translateX(-50%); font-size: 0.75rem; font-weight: 600; color: var(--text-main);">${val}</span>
                            </div>`;
                    });
                }

                // Status (Realizadas, Agendadas, Canceladas)
                const resStatus = await fetch(`http://localhost:8000/api/consultas/estatisticas/status`, {
                    headers: { 'Authorization': `Bearer ${token}` }
                });

                if(resStatus.ok) {
                    const dataStatus = await resStatus.json();
                    document.getElementById('status-realizadas-text').textContent = dataStatus.realizadas + '%';
                    document.getElementById('status-realizadas-bar').style.width = dataStatus.realizadas + '%';

                    document.getElementById('status-agendadas-text').textContent = dataStatus.agendadas + '%';
                    document.getElementById('status-agendadas-bar').style.width = dataStatus.agendadas + '%';

                    document.getElementById('status-canceladas-text').textContent = dataStatus.canceladas + '%';
                    document.getElementById('status-canceladas-bar').style.width = dataStatus.canceladas + '%';

                    document.getElementById('status-nao-realizadas-text').textContent = dataStatus.nao_realizadas + '%';
                    document.getElementById('status-nao-realizadas-bar').style.width = dataStatus.nao_realizadas + '%';
                }

            } catch (error) {
                console.error('Erro gr├ífico:', error);
                chartContainer.innerHTML = '<div style="color: #F56565; font-size: 0.85rem; align-self: center; width: 100%; text-align: center;">Erro ao carregar gr├ífico</div>';
            }
        }

        document.addEventListener('DOMContentLoaded', async () => {
            const token = localStorage.getItem('token');
            if (!token) return;

            const hojeReal = new Date();
            const dataStr = hojeReal.toISOString().split('T')[0];
            let dataBaseCalendario = new Date();

            function renderizarCalendario() {
                const meses = ['Janeiro', 'Fevereiro', 'Mar├ºo', 'Abril', 'Maio', 'Junho', 'Julho', 'Agosto', 'Setembro', 'Outubro', 'Novembro', 'Dezembro'];
                
                document.getElementById('calendar-header').innerHTML = `
                    <span>${meses[dataBaseCalendario.getMonth()]} ${dataBaseCalendario.getFullYear()}</span>
                    <div style="display: flex; gap: 0.5rem;">
                        <i id="cal-prev" data-lucide="chevron-left" style="width: 16px; height: 16px; cursor: pointer;"></i>
                        <i id="cal-next" data-lucide="chevron-right" style="width: 16px; height: 16px; cursor: pointer;"></i>
                    </div>
                `;
                
                document.getElementById('cal-prev').addEventListener('click', () => {
                    dataBaseCalendario.setMonth(dataBaseCalendario.getMonth() - 1);
                    renderizarCalendario();
                });
                document.getElementById('cal-next').addEventListener('click', () => {
                    dataBaseCalendario.setMonth(dataBaseCalendario.getMonth() + 1);
                    renderizarCalendario();
                });

                const grid = document.getElementById('calendar-grid');
                grid.innerHTML = `
                    <span style="color: rgba(255,255,255,0.5); font-size: 0.75rem;">D</span>
                    <span style="color: rgba(255,255,255,0.5); font-size: 0.75rem;">S</span>
                    <span style="color: rgba(255,255,255,0.5); font-size: 0.75rem;">T</span>
                    <span style="color: rgba(255,255,255,0.5); font-size: 0.75rem;">Q</span>
                    <span style="color: rgba(255,255,255,0.5); font-size: 0.75rem;">Q</span>
                    <span style="color: rgba(255,255,255,0.5); font-size: 0.75rem;">S</span>
                    <span style="color: rgba(255,255,255,0.5); font-size: 0.75rem;">S</span>
                `;
                
                const primeiroDia = new Date(dataBaseCalendario.getFullYear(), dataBaseCalendario.getMonth(), 1);
                const ultimoDia = new Date(dataBaseCalendario.getFullYear(), dataBaseCalendario.getMonth() + 1, 0);
                
                // Preenche espa├ºos vazios antes do dia 1
                for(let i = 0; i < primeiroDia.getDay(); i++) {
                    grid.appendChild(document.createElement('span'));
                }
                
                for(let day = 1; day <= ultimoDia.getDate(); day++) {
                    const span = document.createElement('span');
                    span.textContent = day;
                    span.className = 'calendar-day';
                    span.style.cssText = "font-size: 0.85rem; display: flex; align-items: center; justify-content: center; height: 24px; cursor: pointer; transition: 0.2s;";
                    
                    if(day === hojeReal.getDate() && dataBaseCalendario.getMonth() === hojeReal.getMonth() && dataBaseCalendario.getFullYear() === hojeReal.getFullYear()) {
                        span.classList.add('active');
                        span.style.background = "var(--primary)";
                        span.style.color = "white";
                        span.style.borderRadius = "8px";
                        span.style.fontWeight = "700";
                    }
                    
                    span.addEventListener('click', function() {
                        document.querySelectorAll('.calendar-day').forEach(el => {
                            el.classList.remove('active');
                            el.style.background = "transparent";
                            el.style.color = "inherit";
                            el.style.fontWeight = "normal";
                        });
                        
                        this.classList.add('active');
                        this.style.background = "var(--primary)";
                        this.style.color = "white";
                        this.style.borderRadius = "8px";
                        this.style.fontWeight = "700";
                        
                        const m = String(dataBaseCalendario.getMonth() + 1).padStart(2, '0');
                        const d = String(day).padStart(2, '0');
                        const y = dataBaseCalendario.getFullYear();
                        const dStr = `${y}-${m}-${d}`;
                        
                        document.getElementById('agenda-title').textContent = `Agenda: ${d}/${m}/${y}`;
                        carregarAgendaDoDia(dStr);
                    });
                    
                    grid.appendChild(span);
                }
                lucide.createIcons();
            }

            renderizarCalendario();

            // Load Other Stats
            try {
                const me = await window.api.getMe();
                document.getElementById('profile-name').textContent = me.nome;
                document.getElementById('avatar-initial').textContent = me.nome.charAt(0).toUpperCase();

                const userRole = (me.role || '').toUpperCase();
                if (userRole === 'ADMIN' || userRole === 'MEDICO' || userRole === 'RECEPTIONIST') {
                    const pacResponse = await window.api.getPacientes(0, 10);
                    const el = document.getElementById('stat-pacientes');
                    if (el) el.textContent = pacResponse.total;
                }

                if (userRole === 'ADMIN') {
                    const medResponse = await window.api.getMedicos(0, 10);
                    const el = document.getElementById('stat-medicos');
                    if (el) el.textContent = medResponse.total;
                }

            } catch (error) {
                console.error(error);
                showToast('Erro ao carregar dados', 'error');
            }

            // Load Agenda
            carregarAgendaDoDia(dataStr);
            carregarGraficoSemanaEStatus();
        });
